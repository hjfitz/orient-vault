const VERDICTS = {
  type: 'object',
  properties: {
    verdicts: {
      type: 'array',

        type: 'object',

const verifyPrompt = (p, lens, lensText) => `${CONTEXT}

## Your job: adversarially verify the draft page "${p.title}"
Read the draft at ${p.path}. Number every checkable factual claim C1, C2 ... in page order. That covers numbers, named systems and what they do, directions of data flow, who reads or writes what, volumes, dates and negatives such as "nothing" or "only". Include claims inside tables. Skip pure opinion and the "Not confirmed yet" and "Decisions" lists, except check that nothing listed there has in fact been settled.

Your lens is ${lens.toUpperCase()}. ${lensText}

For each claim give one verdict:
- CONFIRMED: you read or measured primary evidence that it is true as written.
- PARTLY: true in substance, but a detail is wrong, stale or overstated. Give the corrected statement.
- REFUTED: primary evidence says otherwise. Give what is true.
- UNVERIFIABLE: you could not reach evidence either way. Say exactly what blocked you.
Watch for absolutes ("only", "nothing", "never", "every", "all"). Overstated wording is PARTLY.`

const judgePrompt = (p, a, b) => `${CONTEXT}

## Your job: judge the claims in draft page "${p.title}"
Two independent verifiers checked the draft at ${p.path}, one through source evidence, one through runtime evidence. Decide a final verdict per claim.

Verifier A (source lens):
${JSON.stringify(a, null, 1)}

Verifier B (runtime lens):
${JSON.stringify(b, null, 1)}

Rules:
- The verifiers number claims independently. Merge them by claim text, not by number, and keep the page order.
- Where they agree and cite primary evidence, accept it.
- Where they disagree, or either says PARTLY, REFUTED or UNVERIFIABLE, check the disputed point yourself against the primary source, and record what you checked in "tiebreak".
- A claim is CONFIRMED only when primary evidence supports every part of it as written.
- Also list any claim that neither verifier covered, and check it yourself.
- For anything not CONFIRMED, write replacement wording for the page: plain British English, 25 words or fewer per sentence, no em dashes, keeping the hedge only where doubt is real.`

const verifyRun = pipeline(
  PAGES,
  p => parallel([
    () => agent(verifyPrompt(p, 'source', p.source), { label: `verify:${p.id}:source`, phase: 'Verify', schema: VERDICTS, model: 'sonnet', effort: 'high' }),
    () => agent(verifyPrompt(p, 'runtime', p.runtime), { label: `verify:${p.id}:runtime`, phase: 'Verify', schema: VERDICTS, model: 'sonnet', effort: 'high' }),
  ]),
  (pair, p) => agent(judgePrompt(p, pair[0], pair[1]), { label: `judge:${p.id}`, phase: 'Judge', schema: JUDGED, effort: 'high' })
    .then(j => (j ? { page: p.id, title: p.title, ...j } : null)),
)

const judged = (await verifyRun).filter(Boolean)
log(`Judged ${judged.length}/${PAGES.length} pages`)

phase('Critic')

Judged verdicts so far:

return { judged, critic }
